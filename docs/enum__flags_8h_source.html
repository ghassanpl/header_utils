324 template <typename FUNC>
325 constexpr auto for_each(FUNC&& callback) const
326 {
327 using return_type = std::invoke_result_t<FUNC, enum_type>;
328 using bitset_type = std::make_unsigned_t<value_type>;
329 auto bitset = static_cast<bitset_type>(bits);
330 while (bitset)
331 {
332#pragma warning(suppress: 4146)
333 const auto t = static_cast<bitset_type>(bitset & -bitset);
334 const auto r = std::countr_zero(t);
335 if constexpr (std::is_convertible_v<return_type, bool>)
336 {
337 if (auto ret = callback((enum_type)r)) return ret;
338 }
339 else
340 callback((enum_type)r);
341 bitset ^= t;
342 }
343 if constexpr (std::is_convertible_v<return_type, bool>)
344 return return_type{};
345 }

349 constexpr void apply(flag_changes changes) noexcept
350 {
351 const auto bits_to_xor = changes.bits_to_set.bits & changes.bits_to_unset.bits;
352 bits ^= bits_to_xor;
353
355 changes.bits_to_set.bits &= ~bits_to_xor;
356 changes.bits_to_unset.bits &= ~bits_to_xor;
357
359 bits &= ~changes.bits_to_unset.bits;
360 bits |= changes.bits_to_set.bits;
361 }
