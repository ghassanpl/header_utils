325 template <typename FUNC>
326 constexpr auto for_each(FUNC&& callback) const
327 {
328 using return_type = std::invoke_result_t<FUNC, enum_type>;
329 using bitset_type = std::make_unsigned_t<value_type>;
330 auto bitset = static_cast<bitset_type>(bits);
331 while (bitset)
332 {
333#pragma warning(suppress: 4146)
334 const auto t = static_cast<bitset_type>(bitset & -bitset);
335 const auto r = std::countr_zero(t);
336 if constexpr (std::is_convertible_v<return_type, bool>)
337 {
338 if (auto ret = callback((enum_type)r)) return ret;
339 }
340 else
341 callback((enum_type)r);
342 bitset ^= t;
343 }
344 if constexpr (std::is_convertible_v<return_type, bool>)
345 return return_type{};
346 }

350 constexpr void apply(flag_changes changes) noexcept
351 {
352 const auto bits_to_xor = changes.bits_to_set.bits & changes.bits_to_unset.bits;
353 bits ^= bits_to_xor;
354
356 changes.bits_to_set.bits &= ~bits_to_xor;
357 changes.bits_to_unset.bits &= ~bits_to_xor;
358
360 bits &= ~changes.bits_to_unset.bits;
361 bits |= changes.bits_to_set.bits;
362 }
