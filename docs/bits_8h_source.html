55 template <size_t BEGIN, size_t END, bit_integral T>
56 constexpr T extract_bits(T value) {
57 static_assert(END > BEGIN, "range size must be greater than zero");
58 static_assert(END <= bit_count<T>, "range must be within type bit size");
59 return T(value >> BEGIN) & T((T(1)<<(END-BEGIN))-1);
60 };

62 template <size_t... RANGES, bit_integral T>
63 constexpr auto split_bit_ranges(T value) {
64 static_assert((RANGES + ...) == bit_count<T>);
65 uint64_t bits = static_cast<uint64_t>(value);
66 auto extract = [&]<size_t N>(std::integral_constant<size_t, N>) {
67 auto result = extract_bits<0, N>(bits);
68 bits >>= N;
69 return result;
70 };
71 return std::array {
72 extract(std::integral_constant<size_t, RANGES>{})...
73 };
74 };

132 constexpr auto cat_bits(BIT_TYPES... bits)
133 {
134 using sum_type = uintN_t<(bit_count<BIT_TYPES> + ...)>;
135 sum_type result = 0;

137 result <<= bit_count<T>;
138 result |= value;

308 template <bit_integral VALUE_TYPE, size_t BIT_NUM = dynamic_bit_number>
309 struct bit_reference : public detail::bit_num_base<VALUE_TYPE, BIT_NUM>
310 {
311 using base_type = detail::bit_num_base<VALUE_TYPE, BIT_NUM>;
312 static constexpr size_t value_bit_count = bit_count<VALUE_TYPE>;
313
314 static constexpr bool is_static = BIT_NUM != dynamic_bit_number;
315 static constexpr bool is_const = std::is_const_v<VALUE_TYPE>;
316
317 constexpr bit_reference(VALUE_TYPE& ref, size_t bitnum) requires (!is_static)
318 : base_type(bitnum)
319 , m_value_ref(&ref)
320 {
321 if (bitnum >= value_bit_count)
322 throw std::invalid_argument("bit_num can't be greater than or equal to the number of bits in the value type");
323 }

325 constexpr bit_reference(VALUE_TYPE& ref, detail::bit_num_t<BIT_NUM> = {}) noexcept requires is_static

329 }
