226 [[nodiscard]] constexpr color_t to_rgb(color_hsva_t const& hsva)
227 {
228 const auto hue = hsva->r;
229 const auto saturation = hsva->g;
230 const auto value = hsva->b;
231 const auto alpha = hsva->a;
232 const auto i = (int)hue;
233 float fraction = hue - (float)i;
234 if (!(i & 1)) fraction = 1.0f - fraction;
235 const auto m = value * (1.0f - saturation);
236 const auto n = value * (1.0f - saturation * fraction);
237 switch (i)
238 {
239 case 6:
240 case 0: return color_t(value, n, m, alpha);
241 case 1: return color_t(n, value, m, alpha);
242 case 2: return color_t(m, value, n, alpha);
243 case 3: return color_t(m, n, value, alpha);
244 case 4: return color_t(n, m, value, alpha);
245 case 5: return color_t(value, m, n, alpha);
246 default: return color_t(value, value, value, alpha);
247 }
248 }

251 [[nodiscard]] constexpr color_hsva_t to_hsv(color_t const& rgba)
252 {
253 const auto min = detail::min(rgba.x, rgba.y, rgba.z);
254 const auto max = detail::max(rgba.x, rgba.y, rgba.z);
255 const auto delta = max - min;
256
257 float h = 0;
258
259 if (delta != 0)
260 {
261 if (rgba.x == max)
262 h = fmod((rgba.y - rgba.z) / delta, 6.0f);
263 else if (rgba.y == max)
264 h = 2 + (rgba.z - rgba.x) / delta;
265 else
266 h = 4 + (rgba.x - rgba.y) / delta;
267 }
268
269 return color_hsva_t{
270 h,
271 (max != 0) ? (delta / max) : 0,
272 max,
273 rgba.w
274 };
275 }

278 [[nodiscard]] constexpr color_rgba_t from_html(const char* str, size_t n)
279 {
280 if (n == 0)
281 throw n;
282 if (str[0] == '#')
283 {
284 str++;
285 n--;
286 }
287 uint8_t r{};
288 uint8_t g{};
289 uint8_t b{};
290 uint8_t a = 255;
291 switch (n)
292 {
293 case 4:
294 std::from_chars(str + 3, str + 4, a, 16); a = a << 4 | a;
295 [[fallthrough]];
296 case 3:
297 std::from_chars(str + 0, str + 1, r, 16); r = r << 4 | r;
298 std::from_chars(str + 1, str + 2, g, 16); g = g << 4 | g;
299 std::from_chars(str + 2, str + 3, b, 16); b = b << 4 | b;
300 break;
301 case 8:
302 std::from_chars(str + 6, str + 8, a, 16);
303 [[fallthrough]];
304 case 6:
305 std::from_chars(str + 0, str + 2, r, 16);
306 std::from_chars(str + 2, str + 4, g, 16);
307 std::from_chars(str + 4, str + 6, b, 16);
308 break;
309 default: throw "invalid number of characters";
310 }
311 return { r / 255.0f, g / 255.0f, b / 255.0f, a / 255.0f };
312 }
