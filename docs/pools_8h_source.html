17 template <typename T, size_t BLOCK_SIZE = 1024>
18 struct pool
19 {
20 template <typename... ARGS>
21 T* create(ARGS&&... args)
22 {
23 if (mPointers.empty())
24 {
25 auto& new_block = mBlocks.emplace_back(std::make_unique<mem_proxy[]>(BLOCK_SIZE));
26 for (size_t i = 0; i < BLOCK_SIZE; ++i)
27 mPointers.push_back(&new_block[i]);
28 }
29 auto result = std::construct_at(reinterpret_cast<T*>(mPointers.back()), std::forward<ARGS>(args)...);
30 mPointers.pop_back();
31 return result;
32 }
33
34 void destroy(T* ptr)
35 {
36 std::destroy_at(ptr);
37 mPointers.push_back(std::launder(reinterpret_cast<mem_proxy*>(ptr)));
38 }
39
40 void clear()
41 {
42 mPointers.clear();
43 mBlocks.clear();
44 }
45
46 size_t capacity() const noexcept { return mBlocks.size() * BLOCK_SIZE; }
47 size_t capacity_in_bytes() const noexcept { return capacity() * sizeof(mem_proxy); }
48 size_t free_elements() const noexcept { return mPointers.size(); }
49 size_t allocated_elements() const noexcept { return capacity() - free_elements(); }
50
51 private:
52
54 struct alignas(alignof(T)) mem_proxy { unsigned char mem[sizeof(T)]; };
55 std::vector<mem_proxy*> mPointers;
56 std::vector<std::unique_ptr<mem_proxy[]>> mBlocks;
57 };
