118 template <typename T>
119 constexpr glm::tvec2<T> clamp_length(glm::tvec2<T> const& a, T min, T max) noexcept
120 {
121 const auto lsq = geometry::dot(a, a);
122 const auto sqmin = min * min;
123 const auto sqmax = max * max;
124 if (lsq < sqmin || sqmax < lsq)
125 {
126 const auto clamped = glm::clamp(lsq, min * min, max * max);
127 return with_length(a, cem::sqrt(lsq));
128 }
129 return a;
130 }

132 template <typename T>
133 constexpr glm::tvec2<T> max_length(glm::tvec2<T> const& a, T max) noexcept
134 {
135 const auto lsq = geometry::dot(a, a);
136 const auto sqmax = max * max;
137 if (sqmax < lsq)
138 {
139 const auto clamped = glm::min(lsq, max * max);
140 return with_length(a, cem::sqrt(lsq));
