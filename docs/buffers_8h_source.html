39 template <typename BUFFER, typename T>
40 bool buffer_append(BUFFER&& buffer, T&& val)
41 {
42 if constexpr (requires { buffer.append(val); })
43 {
44 buffer.append(val);
45 return true;
46 }
47 else if constexpr (requires { buffer.push_back(val); })
48 {
49 buffer.push_back(val);
50 return true;
51 }
52 else if constexpr (requires { buffer.insert(buffer.end(), val); })
53 {
54 buffer.insert(buffer.end(), val);
55 return true;
56 }
57 else if constexpr (requires { buffer.put(val); })
58 {
59 buffer.put(val);
60 return true;
61 }
62 else
63 {
64 static_assert(!always_false<T>, "buffer cannot be appended with this value type - buffer_append might need to be specialized");
65 return false;
66 }

89 template <typename BUFFER, typename RANGE>
90 requires output_buffer<BUFFER, std::ranges::range_value_t<RANGE>>
91 size_t buffer_append_range(BUFFER& buffer, RANGE&& range)
92 {
93 if constexpr (std::ranges::sized_range<RANGE>)
94 buffer_reserve(buffer, std::ranges::size(range));
95
96 if constexpr (requires { buffer.append(std::to_address(std::ranges::begin(range)), std::ranges::size(range)); } && std::ranges::sized_range<RANGE>)
97 {
98 const auto size = std::ranges::size(range);
99 buffer.append(std::to_address(std::ranges::begin(range)), size);
100 return size;
101 }
102 else if constexpr (requires { buffer.write(std::to_address(std::ranges::begin(range)), std::ranges::size(range)); } && std::ranges::sized_range<RANGE>)
103 {
104 const auto size = std::ranges::size(range);
106 buffer.write(std::to_address(std::ranges::begin(range)), size);
109 return size;
110 }
111 else
112 {
113 size_t count = 0;
114 for (auto&& value : std::forward<RANGE>(range))
115 {
116 if (!buffer_append(buffer, value))
117 break;
118 ++count;
119 }
120 return count;
121 }
122 }

149 template <typename BUFFER, typename ELEMENT_TYPE = buffer_element_type<BUFFER>>
150 requires output_buffer<BUFFER, ELEMENT_TYPE>
151 size_t buffer_append_varint(BUFFER& buffer, std::integral auto oval)
152 {
153 auto val = std::bit_cast<std::make_unsigned_t<decltype(oval)>>(oval);
154 if constexpr (std::is_signed_v<decltype(val)>)
155 val = (oval < 0) ? ((std::bit_cast<std::make_unsigned_t<decltype(oval)>>(-oval) << 1) | 1) : (val << 1);
156 size_t result = 0;
157 while (val >= 128)
158 {
159 result += buffer_append(buffer, uint8_t(0x80 | (val & 0x7f)));
160 val >>= 7;
161 }
162 result += buffer_append(buffer, uint8_t(val));
163 return result;
164 }

167 template <typename BUFFER, typename ELEMENT_TYPE = buffer_element_type<BUFFER>>
168 requires output_buffer<BUFFER, ELEMENT_TYPE>
169 size_t buffer_append_utf8(BUFFER& buffer, char32_t cp)
170 {
171 size_t result = 0;
172 if (cp < 0x80ULL)
173 {
174 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>(cp));
175 }
176 else if (cp < 0x800ULL) {
177 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>((cp >> 6) | 0xc0));
178 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>((cp & 0x3f) | 0x80));
179 }
180 else if (cp < 0x10000ULL) {
181 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>((cp >> 12) | 0xe0));
182 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>(((cp >> 6) & 0x3f) | 0x80));
183 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>((cp & 0x3f) | 0x80));
184 }
185 else {
186 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>((cp >> 18) | 0xf0));
187 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>(((cp >> 12) & 0x3f) | 0x80));
188 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>(((cp >> 6) & 0x3f) | 0x80));
189 result += buffer_append(buffer, static_cast<ELEMENT_TYPE>((cp & 0x3f) | 0x80));
190 }
191
192 return result;
193 }
