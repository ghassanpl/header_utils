42 template <char open = '[', char close = ']', typename FUNC>
43 [[nodiscard]] std::string interpolate_recursive(std::string_view str, FUNC&& func)
44 {
45 static_assert(std::invocable<FUNC, std::string_view> && std::constructible_from<std::string, std::invoke_result_t<FUNC, std::string_view>>, "function must take a stringable and return a string");
46 std::string result;
47 while (!str.empty())
48 {
49 result += string_ops::consume_until_delim_ex(str, open);
50 if (str.empty()) break;
51 if (string_ops::consume(str, open))
52 result += open;
53 else
54 {
55 size_t opened = 0;
56 auto key = str;
57 while (!str.empty() && (str[0] != close || opened))
58 {
59 if (str[0] == close && opened)
60 opened--;
61 else if (str[0] == open)
62 opened++;
63 str.remove_prefix(1);
64 }
65 key = { key.data(), str.data() };
66 std::ignore = string_ops::consume(str, close);
67
68 result += func(interpolate_recursive<open, close>(key, func));
69 }
70 }
71 return result;
72 }
