37 template <char open = '[', char close = ']', typename FUNC>
38 [[nodiscard]] std::string interpolate_recursive(std::string_view str, FUNC&& func)
39 {
40 static_assert(std::invocable<FUNC, std::string_view> && std::constructible_from<std::string, std::invoke_result_t<FUNC, std::string_view>>, "function must take a stringable and return a string");
41 std::string result;
42 while (!str.empty())
43 {
44 result += string_ops::consume_until_delim_ex(str, open);
45 if (str.empty()) break;
46 if (string_ops::consume(str, open))
47 result += open;
48 else
49 {
50 size_t opened = 0;
51 auto key = str;
52 while (!str.empty() && (str[0] != close || opened))
53 {
54 if (str[0] == close && opened)
55 opened--;
56 else if (str[0] == open)
57 opened++;
58 str.remove_prefix(1);
59 }
60 key = { key.data(), str.data() };
61 std::ignore = string_ops::consume(str, close);
62
63 result += func(interpolate_recursive<open, close>(key, func));
64 }
65 }
66 return result;
