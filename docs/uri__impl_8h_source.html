46 static char parse_pct(std::string_view& str)
47 {
48 uint8_t value = 0;
49 if (auto h1 = consume(str, ascii::isxdigit))
50 value += ascii::xdigit_to_number(h1) * 16;
51 else
52 throw uri_error_code::invalid_percent_encoding;
53 if (auto h2 = consume(str, ascii::isxdigit))
54 value += ascii::xdigit_to_number(h2);
55 else
56 throw uri_error_code::invalid_percent_encoding;
57 return (char)value;
58 }

62 {
63 std::string result{ prefix };
64 do

66 result += consume_while(str, pred);

75 static std::string try_parse_ipv4(std::string_view& str, enum_flags<uri_decompose_flags> const flags)
76 {
77 auto start = str;
78 auto [_, n1] = consume_c_unsigned(str);
79 if (n1 < 0 || n1 > 255) return {};
80 if (!consume(str, '.')) return {};
81 auto [__, n2] = consume_c_unsigned(str);
82 if (n2 < 0 || n2 > 255) return {};
83 if (!consume(str, '.')) return {};
84 auto [___, n3] = consume_c_unsigned(str);
85 if (n3 < 0 || n3 > 255) return {};
86 if (!consume(str, '.')) return {};
87 auto [____, n4] = consume_c_unsigned(str);
88 if (n4 < 0 || n4 > 255) return {};
89
90 return make_string(start.begin(), str.begin());

169 static std::tuple<std::string, std::vector<std::string>> parse_path(bool with_authority, uri_view& uri, enum_flags<uri_decompose_flags> const flags)
170 {
172 auto path = consume_until_any(uri, '?', '#');
173
174 if (with_authority && !path.empty() && path[0] != '/')
175 throw uri_error_code::path_malformed;
176
177 std::vector<std::string> elements;
178
179 if (!path.empty())
180 {
181 uri_view pv = path;
182
183 trim(pv, '/');
184
185 if (!pv.empty())
186 {
187 do
188 {
189 if (flags.contain(uri_decompose_flags::split_path_elements))
190 elements.push_back(parse_segment(pv, flags));
191 else
192 std::ignore = parse_segment(pv, flags);
193 } while (consume(pv, '/'));
194
195 if (!pv.empty())
196 throw uri_error_code::path_malformed;
197 }
198 }
199
200 return { std::string{ path }, std::move(elements) };
201 }

257 uri_expected<decomposed_uri> decompose_uri(uri_view uri, enum_flags<uri_decompose_flags> const flags)
258 {
259 try
260 {
261 decomposed_uri result{};
262 result.decompose_flags = flags;
263
264 result.scheme = detail::parse_scheme(uri, flags);
265
266 if (!string_ops::consume(uri, ':'))
267 return unexpected(uri_error_code::scheme_malformed);
268
269 if (string_ops::consume(uri, "//"))
270 {
271 result.authority = detail::parse_authority(uri, flags);
272
273 auto [user_info, host, port] = detail::parse_authority_elements(result.authority, flags);
274 if (port.empty() && flags.contain(uri_decompose_flags::use_well_known_port_numbers))
275 port = detail::deduce_port_from_scheme(result.scheme);
276 result.user_info = std::move(user_info);
277 result.host = std::move(host);
278 result.port = std::move(port);
279 }
280
281 auto [path, elements] = detail::parse_path(!result.authority.empty(), uri, flags);
282 result.path = std::move(path);
283 result.path_elements = std::move(elements);
284 if (flags.contain(uri_decompose_flags::normalize_path))
285 result.path_elements = result.normalized_path();
286
287 if (string_ops::consume(uri, '?'))
288 {
289 auto [query, elements] = detail::parse_query(uri, flags);
290 result.query = std::move(query);
291 result.query_elements = std::move(elements);
292 }
293
294 if (string_ops::consume(uri, '#'))
295 result.fragment = detail::parse_fragment(uri, flags);
296
297 if (flags.contains_all_of(uri_decompose_flags::lowercase_when_appropriate, uri_decompose_flags::normalize_path))
298 result.canonical_form = true;
299
300 if (flags.contain(uri_decompose_flags::query_known_scheme))
301 {
302 result.known_scheme = query_uri_scheme(result.scheme);
303 if (result.known_scheme)
304 {
305 return result.known_scheme->validate(result)
306 .transform([&]{ return std::move(result); });
307 }
308 }
309
310 return result;
311 }
312 catch (uri_error_code code)
313 {
314 return unexpected(code);
315 }
316 }
