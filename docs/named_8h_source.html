174 template <template <typename...> typename TRAIT_TYPE, typename GIVEN_PACK, typename FIRST_TRAIT, typename... REST>

176 {
177 using type_candidate = std::remove_cvref_t<typename decltype(find_trait_type_impl<TRAIT_TYPE>(GIVEN_PACK{}, trait))::type > ;
178 if constexpr (std::is_same_v<type_candidate, void>)
179 return find_trait_type_traits<TRAIT_TYPE, GIVEN_PACK>(traits...);
