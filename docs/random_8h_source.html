25 void seed(uint64_t j) noexcept
26 {
27 v = 4101842887655102017LL;
28 w = 1;
29 u = j ^ v; (void)this->operator()();
30 v = u; (void)this->operator()();
31 w = v; (void)this->operator()();
32 }

37 [[nodiscard]] uint64_t operator()() noexcept
38 {
39 u = u * 2862933555777941757LL + 7046029254386353087LL;
40 v ^= v >> 17; v ^= v << 31; v ^= v >> 8;
41 w = 4294957665U * (w & 0xffffffff) + (w >> 32);
42 uint64_t x = u ^ (u << 21); x ^= x >> 35; x ^= x << 4;

44 }

174 template <typename RANDOM = std::default_random_engine, typename T>
175 [[nodiscard]] auto in_range(T from, T to, RANDOM& rng = ::ghassanpl::random::default_random_engine)
176 {
177 if constexpr (std::is_enum_v<T>)
178 {
179 if (from >= to) return T{};
180
181 std::uniform_int_distribution<std::underlying_type_t<T>> dist(from, to);
182 return (T)dist(rng);
183 }
184 else if constexpr (std::is_floating_point_v<T>)
185 {
186 if (from >= to) return T{};
187 return in_real_range(from, to, rng);
188 }
189 else if constexpr (is_any_of_v<T, short, int, long, long long, unsigned short, unsigned int, unsigned long, unsigned long long>)
190 {
191 if (from >= to) return T{};
192 return in_integer_range(from, to, rng);
193 }
194 else if constexpr (
195 is_any_of_v<T, unsigned char, signed char, char, wchar_t, char16_t, char32_t>
196#ifdef __cpp_char8_t
197 || std::is_same_v<T, char8_t>
198#endif
199 )
200 {
201 if (from >= to) return T{};
204 using signed_as_T = std::conditional_t<std::is_signed_v<T>, int64_t, uint64_t>;
205 return static_cast<T>(in_integer_range(static_cast<signed_as_T>(from), static_cast<signed_as_T>(to), rng));
206 }
207 else if constexpr (requires{ lerp(from, to, ::ghassanpl::random::percentage<float>(rng)); })
208 {
209 return lerp(from, to, ::ghassanpl::random::percentage<float>(rng));
210 }
211 else if constexpr (requires{ lerp(from, to, ::ghassanpl::random::percentage<double>(rng)); })
212 {
213 return lerp(from, to, ::ghassanpl::random::percentage<double>(rng));
214 }
215 else if constexpr (requires{ mix(from, to, ::ghassanpl::random::percentage<float>(rng)); })
216 {
217 return mix(from, to, ::ghassanpl::random::percentage<float>(rng));
218 }
219 else if constexpr (requires{ mix(from, to, ::ghassanpl::random::percentage<double>(rng)); })
220 {
221 return mix(from, to, ::ghassanpl::random::percentage<double>(rng));

228 [[nodiscard]] constexpr T halton_sequence(size_t index, size_t base = 2)
229 {
230 static_assert(std::is_floating_point_v<T>, "halton_sequence only works on floating point types");
231
232 auto result = T(0);
233 auto fraction = T(1);
234 while (index > 0)
235 {
236 fraction /= base;
237 result += fraction * (index % base);
238 index /= base;
239 }
240 return result;
241 }

351 [[nodiscard]] auto make_bag_randomizer(T& container, RANDOM& rng = ::ghassanpl::random::default_random_engine)
352 {
353 using Iterator = decltype(std::end(container));
354 struct Randomizer
355 {
356 Randomizer(RANDOM& rng, T& container)
357 : mRNG(rng)
358 {
359 for (auto it = std::begin(container); it != std::end(container); ++it)
360 mIterators.push_back(it);
361 mCurrent = mIterators.end();
362 }
363 auto Next() { if (mCurrent == mIterators.end()) { Shuffle(); } return *mCurrent++; }
364 void Shuffle() { std::shuffle(mIterators.begin(), mIterators.end(), mRNG); mCurrent = mIterators.begin(); }
365 private:
366 RANDOM& mRNG;
367 std::vector<Iterator> mIterators;
368 typename std::vector<Iterator>::iterator mCurrent;
369 };
370
371 return Randomizer{ rng, container };
372 }

389 template <typename RANGE, typename FUNC, typename RANDOM>
390 [[nodiscard]] auto iterator_with_probability(RANGE&& range, FUNC&& prob_func, RANDOM& rng = ::ghassanpl::random::default_random_engine)
391 {
392#ifdef __cpp_lib_ranges
393 static_assert(std::ranges::forward_range<RANGE>, "range must be forward range");
394 static_assert(std::is_invocable_v<FUNC, std::ranges::range_reference_t<RANGE>>);
395 static_assert(std::convertible_to<std::invoke_result_t<FUNC, std::ranges::range_reference_t<RANGE>>, double>);
396#endif
397
398 const auto start = std::begin(range);
399 const auto end = std::end(range);
400
401 const auto sum = std::accumulate(start, end, 0.0, [&](auto acc, auto&& item) { return acc + (double)prob_func(item); });
402 auto target = in_real_range(0.0, sum, rng);
403
404 for (auto it = start; it != end; ++it)
405 {
406 const auto weight = (double)prob_func(*it);
407 if (target < weight)
408 return it;
409 target = target - weight;
410 }
411
412 return end;
413 }
