51 template <typename T, typename RANDOM = std::default_random_engine>
52 [[nodiscard]] glm::tvec2<T> point_in(geometry::tellipse<T> const& el, RANDOM& rng = ::ghassanpl::random::default_random_engine)
53 {
54 const auto phi = between(T{}, glm::two_pi<T>(), rng);
55 const auto p = glm::tvec2<T>{ cos(phi), sin(phi) } * glm::sqrt(percentage<T>(rng)) * el.radii * T(0.5);
56 return p + el.center;

59 template <typename T, typename RANDOM = std::default_random_engine>
60 [[nodiscard]] glm::tvec2<T> point_in(geometry::ttriangle<T> const& tr, RANDOM& rng = ::ghassanpl::random::default_random_engine)
61 {
62 const auto r1 = glm::sqrt(percentage<T>(rng));
63 const auto r2 = percentage<T>(rng);
64
65 return (tr.a * (T(1) - r1) + tr.b * (r1 * (T(1) - r2)) + tr.c * (r2 * r1));
66 }

68 template <typename T, typename RANDOM = std::default_random_engine>
69 [[nodiscard]] glm::tvec2<T> point_in(geometry::immutable::tpolygon<T> const& poly, RANDOM& rng = ::ghassanpl::random::default_random_engine)
70 {
71 if (poly.triangles().empty()) return {};
72
73 auto r = between(T{}, poly.calculate_area());
74 size_t i = 0;
75 while (r > 0.0)
76 {
77 if (!poly.has_triangle(i))
78 return {};
79
80 r -= poly.triangle_area(i);
81 if (r <= 0.0)
82 break;
83 i++;
84 }
85
86 return point_in(poly.triangle(i), rng);
87 }

125 {
126 static std::uniform_int_distribution<typename RANDOM::result_type> dist{ 0, 3 };
