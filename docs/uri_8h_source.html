189 struct known_uri_scheme
190 {
191 virtual ~known_uri_scheme() noexcept = default;
192
193 virtual uri_error validate(uri_view uri) const noexcept
194 {
195 auto decomposed = decompose_uri(uri);
196 if (!decomposed) return decomposed.transform([](auto) {});
197
198 return validate(decomposed.value());
199 }
200
201 virtual uri_error validate(decomposed_uri const& decomposed) const noexcept
202 {
203 if (decomposed.scheme != scheme()) return unexpected(uri_error_code::scheme_invalid);
204
205 return validate_authority(decomposed.authority)
206 .and_then([&] { return validate_path(decomposed.path); })
207 .and_then([&] { return validate_query(decomposed.query); })
208 .and_then([&] { return validate_fragment(decomposed.fragment); });
209 }
210
211 virtual std::string_view scheme() const noexcept = 0;
212
213 virtual uri_error validate_authority(std::string_view fragment) const noexcept
214 {
215 return validate_user_info(fragment)
216 .and_then([&] { return validate_host(fragment); })
217 .and_then([&] { return validate_port(fragment); });
218 }
