118 template <bytelike B, size_t N = std::dynamic_extent>
119 [[nodiscard]] bool nth_bit(std::span<B const, N> range, size_t n) noexcept
120 {
121 if (n >= range.size() * CHAR_BIT)
122 return false;
123 const auto u8range = as_u8s(range);
124 const auto byte = u8range[n / CHAR_BIT];
125 return byte & (1 << (n % CHAR_BIT));
126 }

201 template <bytelike B, std::integral T>
202 [[nodiscard]] constexpr auto to_bytelike_array(T value_)
203 {
204 auto value = std::bit_cast<std::make_unsigned_t<T>>(value_);
205 std::array<B, sizeof(T)> result;
206 for (size_t i = 0; i < sizeof(T); ++i)
207 {
208 result[i] = static_cast<B>(value & 0xFF);
209 value >>= 8;
210 }
211 return result;
212 }

224 template <size_t ALIGN, bytelike T, size_t N = std::dynamic_extent>
225 [[nodiscard]] auto align_front_to(std::span<T, N> bytes) noexcept -> align_front_to_result<T>
226 {
227 static_assert(ALIGN >= 1, "Alignment must be greater or equal to 1");
228 if constexpr (ALIGN == 1)
229 {
230 return bytes;
231 }
232 else
233 {
234 void* ptr = const_cast<std::remove_const_t<T>*>(bytes.data());
235 size_t size = bytes.size();
236 ptr = std::align(ALIGN, ALIGN, ptr, size);
237 if (!ptr)
238 return { bytes, {} };
239
240 return { { bytes.data(), bytes.size() - size}, { const_cast<T*>(reinterpret_cast<std::remove_const_t<T>*>(ptr)), size }};
241 }
242 }

253 template <size_t ALIGN, bytelike T, size_t N = std::dynamic_extent>
254 [[nodiscard]] constexpr auto align_back_to(std::span<T, N> bytes) noexcept -> align_back_to_result<T>
255 {
256 static_assert(ALIGN >= 1, "Alignment must be greater or equal to 1");
257 if constexpr (ALIGN == 1)
258 {
259 return bytes;
260 }
261 else
262 {
263 const auto size = bytes.size();
264
265 if (size < ALIGN)
266 return { {}, bytes };
267
268 const auto aligned_size = size - (size % ALIGN);
269 return { bytes.subspan(0, aligned_size), bytes.subspan(aligned_size, size % ALIGN) };
270 }

284 [[nodiscard]] auto align_to(std::span<T, N> bytes) noexcept -> align_to_result<T>
285 {
286 static_assert(ALIGN >= 1, "Alignment must be greater or equal to 1");
287 if constexpr (ALIGN == 1)
288 {
289 return bytes;
290 }
291 else
292 {
293 void* ptr = const_cast<std::remove_const_t<T>*>(bytes.data());
294 size_t size = bytes.size();
295 ptr = std::align(ALIGN, ALIGN, ptr, size);
296 if (!ptr)
297 return { bytes, {}, {} };
298
299 const auto aligned_size = size - (size % ALIGN);
300 return {
301 { bytes.data(), bytes.size() - size},
302 { const_cast<T*>(reinterpret_cast<std::remove_const_t<T>*>(ptr)), aligned_size },
303 { const_cast<T*>(reinterpret_cast<std::remove_const_t<T>*>(ptr)) + aligned_size, size % ALIGN }

310 template <typename TO, bytelike B, size_t N = std::dynamic_extent>
311 [[nodiscard]] auto aligned_span_cast(std::span<B, N> bytes) noexcept -> align_to_result<B, TO>
312 {
313 const auto [prefix, aligned_original, suffix] = align_to<std::max(alignof(TO), sizeof(TO))>(bytes);
314 const std::span<TO> aligned {
315 reinterpret_cast<TO*>(aligned_original.data()),
316 reinterpret_cast<TO*>(aligned_original.data() + aligned_original.size())
317 };
318 return { prefix, aligned, suffix };
