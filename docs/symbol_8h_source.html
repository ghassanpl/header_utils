132 [[nodiscard]] static internal_value_type empty_value() noexcept { return instance().m_empty_string; }
133 [[nodiscard]] static internal_value_type insert(std::string_view val)
134 {
135 if (val.empty())
136 return empty_value();
137
138 auto& values = instance().m_values;
139 if (auto v = values.find(val); v == values.end())
140 return &*values.insert(std::string{ val }).first;
141 else
142 return &*v;
143 }
144 [[nodiscard]] static std::string_view string_for(internal_value_type val) noexcept { return val ? std::string_view{ *val } : std::string_view{}; }
145 [[nodiscard]] static hash_type hash_for(internal_value_type val) noexcept { return std::hash<const void*>{}(val); }
146
147 [[nodiscard]] static std::strong_ordering compare(internal_value_type a, internal_value_type b) noexcept {
148 return (a == b) ? std::strong_ordering::equal : (*a <=> *b);
149 }
